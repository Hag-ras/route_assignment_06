use assignment7

// 1. Create an explicit collection named "books" with validation
db.createCollection("books", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: { bsonType: "string", minLength: 1 },
        author: { bsonType: ["string", "null"] },
        year: { bsonType: ["int", "null"] },
        genres: {
          bsonType: ["array", "null"],
          items: { bsonType: "string" }
        }
      }
    }
  }
})

// 2. Create an implicit collection by inserting data directly into "authors"
db.authors.insertOne({ name: "Author1", nationality: "British" })

// 3. Create a capped collection named "logs" with size limit 1MB
db.createCollection("logs", { capped: true, size: 1048576 })

// 4. Create an index on the books collection for the title field
db.books.createIndex({ title: 1 })

// 5. Insert one document into the books collection
db.books.insertOne({
  title: "Book One",
  author: "Jane Austen",
  year: 1813,
  genres: ["Romance", "Classic"]
})

// 6. Insert multiple documents into the books collection
db.books.insertMany([
  { title: "Future", author: "George Orwell", year: 2020, genres: ["Science Fiction"] },
  { title: "Brave New World", author: "Aldous Huxley", year: 2006, genres: ["Dystopian", "Science Fiction"] },
  { title: "The Great Gatsby", author: "F. Scott Fitzgerald", year: 1925, genres: ["Classic", "Drama"] }
])

// 7. Insert a new log into the logs collection
db.logs.insertOne({ action: "created", bookTitle: "Future", createdAt: new Date() })

// 8. Update the book with title "Future" change the year to 2022
db.books.updateOne({ title: "Future" }, { $set: { year: 2022 } })

// 9. Find a Book with title "Brave New World"
db.books.findOne({ title: "Brave New World" })

// 10. Find all books published between 1990 and 2010
db.books.find({ year: { $gte: 1990, $lte: 2010 } })

// 11. Find books where the genre includes "Science Fiction"
db.books.find({ genres: "Science Fiction" })

// 12. Skip first two books, limit next three, sort by year descending
db.books.find({}).sort({ year: -1 }).skip(2).limit(3)

// 13. Find books where the year field stored as an integer
db.books.find({ year: { $type: "int" } })

// 14. Find all books where genres field does not include Horror or Science Fiction
db.books.find({ $nor: [{ genres: { $in: ["Horror", "Science Fiction"] } }] })

// 15. Delete all books published before 2000
db.books.deleteMany({ year: { $lt: 2000 } })

// 16. Using aggregation functions, filter books published after 2000 and sort by year descending
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $sort: { year: -1 } }
])

// 17. Using aggregation functions, find books after 2000 and project title, author, year
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $project: { _id: 0, title: 1, author: 1, year: 1 } }
])

// 18. Using aggregation functions, break array of genres into separate documents
db.books.aggregate([{ $unwind: "$genres" }])

// 19. Using aggregation functions, join books with logs
db.books.aggregate([
  {
    $lookup: {
      from: "logs",
      localField: "title",
      foreignField: "bookTitle",
      as: "book_details"
    }
  },
  { $match: { book_details: { $ne: [] } } }
])
